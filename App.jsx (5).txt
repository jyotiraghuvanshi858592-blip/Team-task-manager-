const statusMeta = {
  Pending:       { color:C.orange, bg:"rgba(245,158,11,.12)", icon:"⏳" },

function Avatar({ name="?", size=32 }) {
  const cols = ["#5b7fff","#0fd494","#f59e0b","#ff6b6b","#a78bfa","#38bdf8"];
  const c = cols[name.charCodeAt(0) % cols.length];
  return (
    <div style={{ width:size,height:size,borderRadius:"50%",background:c,flexShrink:0,
      display:"flex",alignItems:"center",justifyContent:"center",
      fontSize:size*.38,fontWeight:800,color:"#fff" }}>
      {name.slice(0,2).toUpperCase()}
    </div>
  );
}

function Badge({ status }) {
  const m = statusMeta[status]||{ color:C.muted,bg:"transparent",icon:"•" };
  return (
    <span style={{ fontSize:11,fontWeight:700,padding:"4px 12px",borderRadius:99,
      background:m.bg,color:m.color,border:`1px solid ${m.color}44`,
      letterSpacing:.5,textTransform:"uppercase",display:"inline-flex",alignItems:"center",gap:5 }}>
      {m.icon} {status}
    </span>
  );
}

function Input({ label, type="text", value, onChange, placeholder, required }) {
  const [f,setF]=useState(false);
  return (
    <div style={{ marginBottom:16 }}>
      <label style={{ display:"block",fontSize:11,fontWeight:700,color:C.muted,
        marginBottom:5,letterSpacing:1,textTransform:"uppercase" }}>{label}</label>
      <input type={type} value={value} onChange={e=>onChange(e.target.value)}
        placeholder={placeholder} required={required}
        onFocus={()=>setF(true)} onBlur={()=>setF(false)}
        style={{ width:"100%",padding:"12px 15px",borderRadius:10,
          background:f?C.card:C.surface,border:`1.5px solid ${f?C.accent:C.border}`,
          color:C.text,fontSize:14,outline:"none",transition:"all .2s",
          boxShadow:f?`0 0 0 3px rgba(91,127,255,.2)`:"none" }} />
    </div>
  );
}

function Btn({ children, onClick, type="button", variant="primary", sm, full, disabled }) {
  const [h,setH]=useState(false);
  const v={
    primary:{ bg:h?"#7b97ff":C.accent, color:"#fff", border:"none" },
    ghost:  { bg:h?C.border+"66":"transparent", color:C.muted, border:`1.5px solid ${C.border}` },
    success:{ bg:h?C.green:`${C.green}20`, color:C.green, border:`1.5px solid ${C.green}44` },
    danger: { bg:h?C.red:`${C.red}20`, color:C.red, border:`1.5px solid ${C.red}44` },
  }[variant];
  return (
    <button type={type} onClick={onClick} disabled={disabled}
      onMouseEnter={()=>setH(true)} onMouseLeave={()=>setH(false)}
      style={{ padding:sm?"7px 15px":"12px 24px",borderRadius:10,cursor:disabled?"not-allowed":"pointer",
        fontSize:sm?12:14,fontWeight:700,width:full?"100%":"auto",opacity:disabled?.5:1,
        background:v.bg,color:v.color,border:v.border||"none",transition:"all .2s",
        display:"inline-flex",alignItems:"center",gap:6 }}>
      {children}
    </button>
  );
}

function Alert({ msg, type="error" }) {
  if (!msg) return null;
  const c = type==="error"?C.red:C.green;
  return (
    <div style={{ padding:"10px 14px",borderRadius:9,marginBottom:14,fontSize:13,
      background:`${c}18`,color:c,border:`1px solid ${c}33`,fontWeight:500 }}>
      {type==="error"?"⚠ ":"✓ "}{msg}
    </div>
  );
}

function AddTaskModal({ onClose, onAdd, userId }) {
  const [title,setTitle]=useState("");
  const [assignee,setAssignee]=useState("");
  const [deadline,setDeadline]=useState("");
  const [status,setStatus]=useState("Pending");
  const [err,setErr]=useState("");

    const task = { id:nextTaskId++, title, assignee, deadline, status, userId };
    TASKS = [...TASKS, task];
    onAdd(task);
  };

  return (
    <div style={{ position:"fixed",inset:0,background:"rgba(0,0,0,.8)",display:"flex",
      alignItems:"center",justifyContent:"center",zIndex:999,padding:20,backdropFilter:"blur(8px)" }}>
      <div style={{ background:C.card,borderRadius:22,padding:32,width:"100%",maxWidth:440,
        border:`1px solid ${C.border}`,animation:"popIn .25s ease" }}>
        <div style={{ display:"flex",justifyContent:"space-between",alignItems:"center",marginBottom:24 }}>
          <h3 style={{ fontWeight:800,fontSize:20,color:C.white }}>＋ New Task</h3>
          <button onClick={onClose} style={{ background:"none",border:"none",color:C.muted,cursor:"pointer",fontSize:22 }}>✕</button>
        </div>
        <form onSubmit={submit}>
          <Input label="Task Title" value={title} onChange={setTitle} placeholder="e.g. Design Landing Page" required />
          <Input label="Assign To" value={assignee} onChange={setAssignee} placeholder="e.g. Priya Sharma" required />
          <Input label="Deadline" type="date" value={deadline} onChange={setDeadline} required />
          <div style={{ marginBottom:16 }}>
            <label style={{ display:"block",fontSize:11,fontWeight:700,color:C.muted,marginBottom:5,letterSpacing:1,textTransform:"uppercase" }}>Status</label>

                border:`1.5px solid ${C.border}`,color:C.text,fontSize:14,outline:"none" }}>
              <option>Pending</option>
              <option>In Progress</option>
              <option>Completed</option>
            </select>
          </div>
          <Alert msg={err} />
          <div style={{ display:"flex",gap:10,marginTop:4 }}>

function TaskCard({ task, idx, onUpdate, onDelete }) {
  const [h,setH]=useState(false);
  const overdue = new Date(task.deadline)<new Date() && task.status!=="Completed";

  const changeStatus = (s) => {
    TASKS = TASKS.map(t=>t.id===task.id?{...t,status:s}:t);
    onUpdate({ ...task, status:s });
  };

  const del = () => {
    if (!window.confirm(`Delete "${task.title}"?`)) return;
    TASKS = TASKS.filter(t=>t.id!==task.id);
    onDelete(task.id);
  };

  return (
    <div onMouseEnter={()=>setH(true)} onMouseLeave={()=>setH(false)}
      style={{ background:h?C.card:C.surface,border:`1.5px solid ${h?C.accent+"66":C.border}`,
        borderRadius:14,padding:"16px 20px",display:"flex",alignItems:"center",gap:14,flexWrap:"wrap",
        animation:`slideIn .35s ease ${idx*.05}s both`,transition:"all .2s",
        boxShadow:h?`0 6px 30px rgba(91,127,255,.15)`:"none" }}>
      <div style={{ flex:1,minWidth:180 }}>
        <div style={{ display:"flex",alignItems:"center",gap:8,marginBottom:5,flexWrap:"wrap" }}>
          <span style={{ fontWeight:700,fontSize:15,color:C.white }}>{task.title}</span>
          {overdue && <span style={{ fontSize:10,fontWeight:700,color:C.red,background:`${C.red}18`,padding:"2px 8px",borderRadius:99 }}>OVERDUE</span>}
        </div>
        <div style={{ display:"flex",alignItems:"center",gap:7 }}>
          <Avatar name={task.assignee} size={22} />
          <span style={{ fontSize:13,color:C.muted }}>{task.assignee}</span>
        </div>
      </div>
      <div style={{ textAlign:"center",minWidth:100 }}>
        <div style={{ fontSize:10,color:C.muted,textTransform:"uppercase",letterSpacing:.7,marginBottom:2 }}>Deadline</div>
        <div style={{ fontSize:13,fontWeight:600,color:overdue?C.red:C.text }}>
          {new Date(task.deadline).toLocaleDateString("en-IN",{day:"numeric",month:"short",year:"numeric"})}
        </div>
      </div>
      <Badge status={task.status} />
      <div style={{ display:"flex",gap:6,flexWrap:"wrap" }}>
        {task.status==="Pending" && <Btn sm variant="ghost" onClick={()=>changeStatus("In Progress")}>▶ Start</Btn>}
        {task.status!=="Completed" && <Btn sm variant="success" onClick={()=>changeStatus("Completed")}>✓ Done</Btn>}
        {task.status==="Completed" && <Btn sm variant="ghost" onClick={()=>changeStatus("Pending")}>↩ Redo</Btn>}
        <Btn sm variant="danger" onClick={del}>🗑</Btn>
      </div>
    </div>
  );
}

function Dashboard({ user, onLogout }) {
  const [tasks,setTasks]=useState(TASKS.filter(t=>t.userId===user.id));
  const [filter,setFilter]=useState("All");
  const [search,setSearch]=useState("");
  const [showAdd,setShowAdd]=useState(false);

  const refresh = () => setTasks([...TASKS.filter(t=>t.userId===user.id)]);

  const total=tasks.length, completed=tasks.filter(t=>t.status==="Completed").length,
    inProg=tasks.filter(t=>t.status==="In Progress").length, pending=tasks.filter(t=>t.status==="Pending").length;
  const pct = total?Math.round((completed/total)*100):0;

  const filtered = tasks.filter(t=>
    (filter==="All"||t.status===filter)&&
    (t.title.toLowerCase().includes(search.toLowerCase())||t.assignee.toLowerCase().includes(search.toLowerCase()))
  );

  const stats=[
    { label:"Total Tasks", icon:"📋", val:total,     col:C.accent },
    { label:"Completed",   icon:"✅", val:completed,  col:C.green },
    { label:"In Progress", icon:"⚡", val:inProg,     col:"#38bdf8" },
    { label:"Pending",     icon:"⏳", val:pending,    col:C.orange },
  ];

  return (
    <div style={{ minHeight:"100vh",background:C.bg }}>
      <nav style={{ background:C.surface,borderBottom:`1px solid ${C.border}`,padding:"0 24px",
        display:"flex",alignItems:"center",justifyContent:"space-between",height:60,
        position:"sticky",top:0,zIndex:100 }}>
        <div style={{ display:"flex",alignItems:"center",gap:10 }}>
          <span style={{ fontSize:24 }}>⚡</span>
          <span style={{ fontSize:20,fontWeight:900,color:C.white }}>TaskFlow</span>
        </div>
        <div style={{ display:"flex",alignItems:"center",gap:10 }}>
          <Avatar name={user.name} size={32} />
          <Btn sm variant="ghost" onClick={onLogout}>Logout</Btn>
        </div>
      </nav>

      <div style={{ maxWidth:1080,margin:"0 auto",padding:"28px 20px" }}>
        <div style={{ display:"flex",justifyContent:"space-between",alignItems:"center",marginBottom:28,flexWrap:"wrap",gap:14 }}>
          <div>
            <h2 style={{ fontSize:28,fontWeight:900,color:C.white }}>Dashboard 👋</h2>
            <p style={{ color:C.muted,fontSize:14,marginTop:4 }}>Welcome, <span style={{ color:C.accent,fontWeight:600 }}>{user.name}</span></p>
          </div>
          <Btn onClick={()=>setShowAdd(true)}>＋ Add Task</Btn>
        </div>

        <div style={{ background:C.card,borderRadius:16,padding:"18px 22px",marginBottom:20,border:`1px solid ${C.border}` }}>
          <div style={{ display:"flex",justifyContent:"space-between",marginBottom:10 }}>
            <span style={{ fontWeight:700,fontSize:14 }}>Overall Progress</span>
            <span style={{ fontWeight:800,color:C.green,fontSize:15 }}>{pct}%</span>
          </div>
          <div style={{ background:C.border,borderRadius:99,height:8,overflow:"hidden" }}>
            <div style={{ width:`${pct}%`,height:"100%",borderRadius:99,transition:"width .8s ease",
              background:`linear-gradient(90deg,${C.accent},${C.green})` }} />
          </div>
          <p style={{ fontSize:12,color:C.muted,marginTop:8 }}>{completed} of {total} tasks completed</p>
        </div>

        <div style={{ display:"grid",gridTemplateColumns:"repeat(auto-fit,minmax(180px,1fr))",gap:14,marginBottom:28 }}>
          {stats.map((s,i)=>(
            <div key={i} style={{ background:C.card,borderRadius:14,padding:20,border:`1px solid ${C.border}`,
              borderLeft:`3px solid ${s.col}`,animation:`fadeUp .4s ease ${i*.08}s both` }}>
              <div style={{ fontSize:24,marginBottom:8 }}>{s.icon}</div>
              <div style={{ fontSize:36,fontWeight:900,color:s.col,lineHeight:1 }}>{s.val}</div>
              <div style={{ fontSize:13,color:C.muted,marginTop:4,fontWeight:500 }}>{s.label}</div>
            </div>
          ))}
        </div>

        <div style={{ display:"flex",gap:8,marginBottom:18,flexWrap:"wrap",alignItems:"center" }}>
          {["All","Pending","In Progress","Completed"].map(f=>(
            <button key={f} onClick={()=>setFilter(f)}
              style={{ padding:"7px 16px",borderRadius:99,fontSize:13,fontWeight:700,cursor:"pointer",transition:"all .2s",
                background:filter===f?C.accent:C.surface,color:filter===f?"#fff":C.muted,border:`1.5px solid ${filter===f?C.accent:C.border}` }}>
              {f}
            </button>
          ))}
          <input value={search} onChange={e=>setSearch(e.target.value)} placeholder="🔍 Search…"
            style={{ marginLeft:"auto",padding:"8px 14px",borderRadius:10,background:C.surface,
              border:`1.5px solid ${C.border}`,color:C.text,fontSize:13,outline:"none",width:180 }} />
        </div>

        {filtered.length===0 ? (
          <div style={{ textAlign:"center",padding:"60px 0",color:C.muted }}>
            <div style={{ fontSize:48,marginBottom:12 }}>📭</div>
            <p style={{ fontSize:16,fontWeight:600 }}>No tasks found</p>
          </div>
        ) : (
          <div style={{ display:"flex",flexDirection:"column",gap:10 }}>
            {filtered.map((t,i)=>(
              <TaskCard key={t.id} task={t} idx={i}
                onUpdate={u=>{TASKS=TASKS.map(x=>x.id===u.id?u:x);refresh();}}
                onDelete={id=>{TASKS=TASKS.filter(x=>x.id!==id);refresh();}} />
            ))}
          </div>
        )}
      </div>

      {showAdd && <AddTaskModal userId={user.id} onClose={()=>setShowAdd(false)}
        onAdd={t=>{setTasks(p=>[t,...p]);setShowAdd(false);}} />}
    </div>
  );
}

export default function App() {
  const [page,setPage]=useState("login");
  const [user,setUser]=useState(null);

  useEffect(()=>{
    const el=document.createElement("style");
    el.textContent=STYLE;
    document.head.appendChild(el);
  },[]);

  if (page==="signup") return <SignupPage goLogin={()=>setPage("login")} />;
  if (page==="login")  return <LoginPage onLogin={u=>{setUser(u);setPage("dashboard");}} goSignup={()=>setPage("signup")} />;
  return <Dashboard user={user} onLogout={()=>{setUser(null);setPage("login");}} />;
}
